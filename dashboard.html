int targetSector2 = -1; int target2Intensity = 0; int target2Class = TARGET_CLEAR; float target2Speed = 0.0; if (numberOfTargets == 2) { targetSector2 = random(0, NUM_SECTORS); while (abs(targetSector2 - targetSector1) < 4) { targetSector2 = random(0, NUM_SECTORS); int servoAngle = angle % 180; radarServo.write(servoAngle); delay(100); int intensity = random(0, 11); float frequency = random(0, 100); if (i == targetSector1) { intensity = target1Intensity; frequency = random(200, 2001); } else if ( i == targetSector1 - 1 || i == targetSector1 + 1 ) { intensity = random(15, 35); frequency = random(200, 800); } if (i == targetSector2) { intensity = target2Intensity; frequency = random(200, 2001); } else if ( i == targetSector2 - 1 || i == targetSector2 + 1 ) { intensity = random(15, 35); frequency = random(200, 800); } sectorIntensity[i] = intensity; sectorFrequency[i] = frequency; Serial.print("Scanning Sector: ");
